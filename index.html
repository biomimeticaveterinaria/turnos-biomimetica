async function actualizarDisponibilidad() { const fecha = document.getElementById('fechaTurno').value; if (!fecha) return; // 1. Limpiar selecciones previas document.querySelectorAll('.time-slot').forEach(slot => { slot.classList.remove('disabled', 'selected'); }); selectedHora = ''; document.getElementById('horarioSeleccionado').value = ''; // 2. Definir rango del día completo const timeMin = new Date(`${fecha}T00:00:00`).toISOString(); const timeMax = new Date(`${fecha}T23:59:59`).toISOString(); const url = `https://www.googleapis.com/calendar/v3/calendars/${encodeURIComponent(CALENDAR_ID)}/events?key=${API_KEY}&timeMin=${encodeURIComponent(timeMin)}&timeMax=${encodeURIComponent(timeMax)}&singleEvents=true`; try { const response = await fetch(url); const data = await response.json(); if (data.items && data.items.length > 0) { data.items.forEach(event => { if (event.start) { // Extraer fecha/hora de inicio const startString = event.start.dateTime || event.start.date; if (startString.includes('T')) { const dateObj = new Date(startString); // Formatear hora en formato HH:mm local const horas = String(dateObj.getHours()).padStart(2, '0'); const minutos = String(dateObj.getMinutes()).padStart(2, '0'); const horaFormateada = `${horas}:${minutos}`; // Deshabilitar el botón correspondiente const slot = document.querySelector(`.time-slot[data-time="${horaFormateada}"]`); if (slot) { slot.classList.add('disabled'); } } } }); } } catch (e) { console.error("Error al verificar disponibilidad:", e); } }
